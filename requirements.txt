# --- ROCm (从官方索引装, 不从 PyPI) ---
# 索引: https://stable.repo.amd.com/rocm/whl-next/
# gfx1150 = Radeon 880M (Strix Point / Ryzen AI 9 365)
#
# torch==2.12.0+rocm10.0.0   cp314 wheel 已在索引里 (已用 uv --dry-run 验证可解析)
#
# ✗ 不装 torchvision / torchaudio —— 纯文本项目用不到, 省几个 GB
# ✗ 不装 deepspeed / wandb
#   deepspeed 是 ZeRO (多卡参数切片) —— 本项目单卡小模型, 没有多卡可切,
#   收益 0。而且它在 ROCm + Python 3.14 上大概率编译失败。
#
# accelerate 要装。它看着像「分布式训练才用得上」, 但 **peft 硬依赖它**
# (peft 0.17.1 要求 accelerate>=0.21.0), 装 peft 就会把它拉进来。
# 它本身是纯 Python, 不带 deepspeed, 装了没有代价。不显式装的话,
# uv 解析时会顺手把整个 torch 重新解一遍。

# --- HF 栈 ---
# transformers 钉 4.57.0 是 dllm 的 pin, 不是随手选的:
#   dllm/pipelines/a2d/models/qwen3/modeling_qwen3.py 照 4.57 的
#   create_causal_mask 写, 5.x 改了签名, 接不上。
# 4.57.0 在 PyPI 上是 yank 状态 (打包 setup 出错), wheel 本身可用, 已实测。
transformers==4.57.0
peft==0.17.1            # T1a 的 LoRA / SFT
accelerate              # peft 的硬依赖, 不要试图跳过
datasets==4.2.0
tokenizers              # 独立 Rust 库, 读 tokenizer.json 用
safetensors             # 读 HF 权重
numpy
pyyaml                  # configs/*.yaml
tqdm
